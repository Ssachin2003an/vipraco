# ------------------ Authentication ------------------ #
def hash_password(password: str) -> str:
    return bcrypt.hashpw(password.encode('utf-8'), bcrypt.gensalt()).decode('utf-8')

def verify_password(password: str, hashed: str) -> bool:
    return bcrypt.checkpw(password.encode('utf-8'), hashed.encode('utf-8'))

def create_access_token(data: dict) -> str:
    return jwt.encode(data, JWT_SECRET, algorithm=JWT_ALGORITHM)

def verify_token(credentials: HTTPAuthorizationCredentials = Depends(security)) -> dict:
    try:
        payload = jwt.decode(credentials.credentials, JWT_SECRET, algorithms=[JWT_ALGORITHM])
        return payload
    except jwt.PyJWTError:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid authentication credentials")

# ------------------ Utility ------------------ #
def serialize_mongo_doc(doc):
    if doc is None:
        return None
    if "_id" in doc:
        del doc["_id"]
    for key, value in doc.items():
        if isinstance(value, datetime):
            doc[key] = value.isoformat()
        elif isinstance(value, ObjectId):
            doc[key] = str(value)
    return doc

# ------------------ HR Query Logic ------------------ #
async def process_hr_query(user_id: str, organization_id: str, query: str) -> Dict[str, Any]:
    query_lower = query.lower()
    user = await db.users.find_one({"user_id": user_id, "organization_id": organization_id})
    if not user:
        return {"type": "error", "response": "User not found"}

    # Joining date
    if any(k in query_lower for k in ["joining date", "joined", "start date"]):
        joining_date = user["date_of_joining"].strftime("%B %d, %Y")
        return {"type": "hr_data", "response": f"You joined the company on {joining_date}"}

    # Contract information
    if any(k in query_lower for k in ["contract", "agreement", "bond", "tenure"]):
        years = user.get("contract_years")
        end_date = user.get("contract_end_date")
        if not years or not end_date:
            return {"type": "hr_data", "response": "I couldn't find your contract details in the system."}
        if isinstance(end_date, str):
            try:
                end_date = datetime.fromisoformat(end_date)
            except ValueError:
                return {"type": "hr_data", "response": "Invalid contract date format."}

        end_date_str = end_date.strftime("%B %d, %Y")
        joining_date = user["date_of_joining"].strftime("%B %d, %Y")
        today = datetime.now()
        completed_years = (today - user["date_of_joining"]).days // 365
        remaining_years = max(years - completed_years, 0)
        response = (
            f"Your contract period is {years} years. You joined on {joining_date}, "
            f"and your contract ends on {end_date_str}. You have completed about "
            f"{completed_years} year(s) and have approximately {remaining_years} year(s) remaining."
        )
        return {"type": "hr_data", "response": response}

    # Leave balance
    if "leave" in query_lower:
        leaves = await db.leave_balances.find({"user_id": user_id}).to_list(100)
        if not leaves:
            return {"type": "hr_data", "response": "No leave balance information found."}
        summary = [f"{lv['leave_type']}: {lv['total_allotted'] - lv['leaves_taken']} remaining" for lv in leaves]
        return {"type": "hr_data", "response": "\n".join(summary)}

    # Payroll
    if "salary" in query_lower or "ctc" in query_lower:
        payroll = await db.payroll_data.find_one({"user_id": user_id})
        if payroll:
            response = f"Your total CTC is ₹{payroll['ctc']:,.0f} with base salary ₹{payroll['base_salary']:,.0f}."
            return {"type": "hr_data", "response": response}

    # Policies
    if "policy" in query_lower or "policies" in query_lower:
        policies = await db.company_policies.find({"organization_id": organization_id}).to_list(100)
        if not policies:
            return {"type": "hr_data", "response": "No policies found for your organization."}
        titles = [p["policy_title"] for p in policies]
        return {"type": "hr_data", "response": "Available policies:\n" + "\n".join(titles)}

    return None

# ------------------ API Endpoints ------------------ #
@api_router.post("/login", response_model=LoginResponse)
async def login(req: LoginRequest):
    user = await db.users.find_one({"email": req.email})
    if not user or not verify_password(req.password, user["password_hash"]):
        raise HTTPException(status_code=401, detail="Invalid credentials")
    org = await db.organizations.find_one({"organization_id": user["organization_id"]})
    token = create_access_token({"user_id": user["user_id"], "organization_id": user["organization_id"]})
    return LoginResponse(token=token, user=serialize_mongo_doc({k: v for k, v in user.items() if k != "password_hash"}), organization=serialize_mongo_doc(org))

@api_router.post("/chat", response_model=ChatResponse)
async def chat(message: ChatMessage, current_user: dict = Depends(verify_token)):
    user_id, org_id = current_user["user_id"], current_user["organization_id"]
    hr_response = await process_hr_query(user_id, org_id, message.message)
    if hr_response:
        return ChatResponse(response=hr_response["response"], type=hr_response["type"])
    try:
        user = await db.users.find_one({"user_id": user_id})
        org = await db.organizations.find_one({"organization_id": org_id})
        prompt = f"You are VipraCo, AI assistant for {org['org_name']}. {user['first_name']} asked: {message.message}"
        res = model.generate_content(prompt)
        return ChatResponse(response=res.text, type="general_ai")
    except Exception as e:
        logging.error(f"Gemini API error: {e}")
        return ChatResponse(response="I'm sorry, something went wrong.", type="error")
